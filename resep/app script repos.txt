// Ganti dengan ID Spreadsheet Anda (Ambil dari URL Spreadsheet)
const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId(); 

// SKEMA DATABASE (AUTO MAKER)
const DB_SCHEMA = {
  'users': ['id', 'email', 'password', 'nama', 'role', 'avatar', 'created_at', 'updated_at'],
  'settings': ['key', 'value'],
  'ukurans': ['id', 'nama', 'simbol', 'created_at', 'updated_at'],
  'bahans': ['id', 'nama', 'jumlah_dasar', 'ukuran_id', 'harga_beli', 'created_at', 'updated_at'],
  'reseps': ['id', 'nama', 'porsi_dasar', 'cara_masak', 'created_at', 'updated_at'],
  'resep_bahans': ['id', 'resep_id', 'bahan_id', 'jumlah_dibutuhkan', 'created_at', 'updated_at'],
  'konversis': ['id', 'kategori', 'dari_ukuran_id', 'ke_ukuran_id', 'pengali', 'catatan', 'created_at', 'updated_at'],
  'menus': ['id', 'resep_id', 'kategori', 'harga_jual', 'deskripsi', 'gambar', 'status_aktif', 'created_at', 'updated_at'],
  'penjualan': ['id', 'kode_penjualan', 'menu_id', 'jumlah_porsi', 'total_modal', 'total_pendapatan', 'total_keuntungan', 'created_at']
};

// Fungsi ini bisa dijalankan manual dari editor GAS untuk Generate Semua Sheet sekaligus
function setupDatabase() {
  for (let sheetName in DB_SCHEMA) {
    getSheet(sheetName);
  }
  return "Database berhasil di-setup!";
}

function doPost(e) {
  let output = ContentService.createTextOutput();
  output.setMimeType(ContentService.MimeType.JSON);
  
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;
    let result = {};

    switch(action) {
      case 'login':
        result = doLogin(payload.data);
        break;
      case 'readAll':
        result = readAll();
        break;
      case 'insert':
        result = insertData(payload.table, payload.data);
        break;
      case 'update':
        result = updateData(payload.table, payload.id, payload.data);
        break;
      case 'delete':
        result = deleteData(payload.table, payload.id);
        break;
      case 'insertMultiple':
        result = insertMultiple(payload.table, payload.data);
        break;
      case 'saveSettings':
        result = saveSettings(payload.data);
        break;
      default:
        throw new Error("Action tidak dikenali!");
    }

    result.status = 'success';
    output.setContent(JSON.stringify(result));
    
  } catch (error) {
    output.setContent(JSON.stringify({ status: 'error', message: error.toString() }));
  }
  
  return output;
}

// Menangani Pre-flight CORS request
function doOptions(e) {
  let output = ContentService.createTextOutput();
  output.setMimeType(ContentService.MimeType.JSON);
  output.setContent(JSON.stringify({ status: 'ok' }));
  return output;
}

function getSheet(sheetName) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  
  // Jika sheet belum ada, buat otomatis
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    
    // Auto-build Kolom Header berdasarkan Skema
    if (DB_SCHEMA[sheetName]) {
      // Masukkan header baris pertama
      sheet.appendRow(DB_SCHEMA[sheetName]);
      
      // Mempercantik header (Bold & Warna Background Hijau Terang)
      sheet.getRange(1, 1, 1, DB_SCHEMA[sheetName].length)
           .setFontWeight("bold")
           .setBackground("#d9ead3");
           
      // Auto-inject Akun Admin Default jika membuat tabel users
      if (sheetName === 'users') {
        sheet.appendRow([
          new Date().getTime(), 
          'admin@resep.com', 
          'admin123', 
          'Super Admin', 
          'admin', 
          '', 
          new Date().toISOString(), 
          new Date().toISOString()
        ]);
      }
    }
  }
  return sheet;
}

function sheetToJSON(sheet) {
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return []; // Hanya header
  
  const headers = data[0];
  const result = [];
  
  for (let i = 1; i < data.length; i++) {
    let obj = {};
    for (let j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return result;
}

// ==========================================
// FUNGSI AUTHENTICATION
// ==========================================
function doLogin(data) {
  const sheet = getSheet('users');
  const users = sheetToJSON(sheet);
  
  const user = users.find(u => u.email === data.email && u.password === data.password);
  if (!user) throw new Error("Email atau Password salah!");
  
  // Hapus password dari response demi keamanan
  delete user.password;
  
  return { user: user };
}

// ==========================================
// FUNGSI READ ALL DATA & SETTINGS
// ==========================================
function readAll() {
  const response = {
    ukurans: sheetToJSON(getSheet('ukurans')),
    bahans: sheetToJSON(getSheet('bahans')),
    reseps: sheetToJSON(getSheet('reseps')),
    resep_bahans: sheetToJSON(getSheet('resep_bahans')),
    konversis: sheetToJSON(getSheet('konversis')),
    menus: sheetToJSON(getSheet('menus')),
    penjualan: sheetToJSON(getSheet('penjualan')),
    settings: {}
  };

  // Baca Settings
  const settingSheet = getSheet('settings');
  const setValues = settingSheet.getDataRange().getValues();
  if (setValues.length > 1) {
    for (let i = 1; i < setValues.length; i++) {
      const key = setValues[i][0];
      const val = setValues[i][1];
      if(key) response.settings[key] = val;
    }
  }

  return response;
}

// ==========================================
// FUNGSI SIMPAN PENGATURAN (SETTINGS)
// ==========================================
function saveSettings(settingsObj) {
  const sheet = getSheet('settings');
  const data = sheet.getDataRange().getValues();
  
  // Setup header jika kosong
  if (data.length === 0 || data[0][0] !== 'key') {
    sheet.getRange(1, 1, 1, 2).setValues([['key', 'value']]);
  }

  // Map key ke baris (row index)
  let keyRowMap = {};
  for (let i = 1; i < data.length; i++) {
    keyRowMap[data[i][0]] = i + 1; 
  }

  for (const [key, value] of Object.entries(settingsObj)) {
    if (keyRowMap[key]) {
      // Update nilai jika key sudah ada
      sheet.getRange(keyRowMap[key], 2).setValue(value);
    } else {
      // Tambah baris baru jika key belum ada
      sheet.appendRow([key, value]);
    }
  }
  
  return { message: "Pengaturan berhasil disimpan ke Server" };
}

// ==========================================
// FUNGSI CRUD STANDAR
// ==========================================
function insertData(table, data) {
  const sheet = getSheet(table);
  const headers = sheet.getDataRange().getValues()[0] || Object.keys(data);
  
  // Set headers jika sheet baru
  if (sheet.getLastRow() === 0) {
    headers.unshift('id'); 
    sheet.appendRow(headers);
  }

  // Generate ID baru
  const newId = new Date().getTime();
  data.id = newId;

  const rowData = headers.map(header => data[header] !== undefined ? data[header] : '');
  sheet.appendRow(rowData);
  
  return { id: newId, message: "Data berhasil ditambahkan" };
}

function updateData(table, id, data) {
  const sheet = getSheet(table);
  const sheetData = sheet.getDataRange().getValues();
  const headers = sheetData[0];
  
  let rowIndex = -1;
  for (let i = 1; i < sheetData.length; i++) {
    if (sheetData[i][0] == id) { rowIndex = i + 1; break; }
  }
  
  if (rowIndex === -1) throw new Error("Data tidak ditemukan untuk diupdate!");
  
  for (let key in data) {
    const colIndex = headers.indexOf(key);
    if (colIndex > -1) {
      sheet.getRange(rowIndex, colIndex + 1).setValue(data[key]);
    }
  }
  
  return { message: "Data berhasil diperbarui" };
}

function deleteData(table, id) {
  const sheet = getSheet(table);
  const sheetData = sheet.getDataRange().getValues();
  
  let rowIndex = -1;
  for (let i = 1; i < sheetData.length; i++) {
    if (sheetData[i][0] == id) { rowIndex = i + 1; break; }
  }
  
  if (rowIndex === -1) throw new Error("Data tidak ditemukan untuk dihapus!");
  
  sheet.deleteRow(rowIndex);
  return { message: "Data berhasil dihapus" };
}

function insertMultiple(table, dataArray) {
  const sheet = getSheet(table);
  let headers = sheet.getDataRange().getValues()[0];
  
  if (!headers) {
    headers = ['id', ...Object.keys(dataArray[0])];
    sheet.appendRow(headers);
  }

  const rowsToInsert = dataArray.map((data, index) => {
    const newId = new Date().getTime() + index;
    data.id = newId;
    return headers.map(header => data[header] !== undefined ? data[header] : '');
  });
  
  if(rowsToInsert.length > 0) {
    sheet.getRange(sheet.getLastRow() + 1, 1, rowsToInsert.length, headers.length).setValues(rowsToInsert);
  }
  
  return { message: `${rowsToInsert.length} data berhasil ditambahkan` };
}
