function doPost(e) {
  try {
    const request = JSON.parse(e.postData.contents);
    const action = request.action;
    const tableName = request.table;
    
    // Autentikasi / Login
    if (action === 'login') {
      const { email, password } = request.data;
      const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('users');
      if (!sheet) throw new Error("Tabel users tidak ditemukan.");
      
      const data = sheet.getDataRange().getValues();
      const headers = data[0];
      
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const record = {};
        headers.forEach((h, index) => record[h] = row[index]);
        
        if (record.email === email && record.password === password) {
          // Jangan kembalikan password ke frontend untuk keamanan
          delete record.password; 
          return responseJSON({ status: 'success', user: record });
        }
      }
      throw new Error("Email atau password salah!");
    }

    if (action === 'readAll') {
      return responseJSON(readAllData());
    }

    if (!tableName) throw new Error("Nama tabel (table) wajib disertakan!");

    if (action === 'insert') {
      return responseJSON(insertData(tableName, request.data));
    } else if (action === 'update') {
      return responseJSON(updateData(tableName, request.id, request.data));
    } else if (action === 'delete') {
      return responseJSON(deleteData(tableName, request.id));
    } else if (action === 'insertMultiple') {
      return responseJSON(insertMultipleData(tableName, request.data));
    } else {
      throw new Error("Aksi tidak dikenal.");
    }
  } catch (error) {
    return responseJSON({ status: 'error', message: error.message });
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function forceCleanPhantomRows() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheets = ss.getSheets();
  
  sheets.forEach(sheet => {
    const maxRows = sheet.getMaxRows();
    const lastRow = sheet.getLastRow();
    if (maxRows > lastRow) {
      sheet.deleteRows(lastRow + 1, maxRows - lastRow);
    }
  });
  Logger.log("Pembersihan baris kosong selesai!");
}

function generateId() {
  return new Date().getTime() + Math.floor(Math.random() * 1000);
}

function readAllData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheets = ss.getSheets();
  const result = { status: 'success' };
  
  sheets.forEach(sheet => {
    const name = sheet.getName();
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      result[name] = [];
      return;
    }
    const headers = data[0];
    const rows = [];
    for (let i = 1; i < data.length; i++) {
      let obj = {};
      let isEmptyRow = true;
      headers.forEach((header, index) => {
        obj[header] = data[i][index];
        if (String(data[i][index]).trim() !== '') isEmptyRow = false;
      });
      if (!isEmptyRow) rows.push(obj);
    }
    result[name] = rows;
  });
  
  return result;
}

function insertData(tableName, dataObj) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tableName);
  if (!sheet) throw new Error("Tabel tidak ditemukan: " + tableName);
  
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  dataObj.id = generateId();
  if(!dataObj.created_at) dataObj.created_at = new Date().toISOString();
  
  const rowData = headers.map(header => dataObj[header] !== undefined ? dataObj[header] : "");
  sheet.appendRow(rowData);
  return { status: 'success', message: 'Data berhasil ditambahkan', id: dataObj.id };
}

function insertMultipleData(tableName, dataArray) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tableName);
  if (!sheet) throw new Error("Tabel tidak ditemukan: " + tableName);
  
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const rowsData = [];
  
  dataArray.forEach(dataObj => {
    dataObj.id = generateId();
    if(!dataObj.created_at) dataObj.created_at = new Date().toISOString();
    const rowData = headers.map(header => dataObj[header] !== undefined ? dataObj[header] : "");
    rowsData.push(rowData);
  });
  
  sheet.getRange(sheet.getLastRow() + 1, 1, rowsData.length, headers.length).setValues(rowsData);
  return { status: 'success', message: rowsData.length + ' Data berhasil ditambahkan' };
}

function updateData(tableName, id, dataObj) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tableName);
  if (!sheet) throw new Error("Tabel tidak ditemukan: " + tableName);
  
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  let rowIndex = -1;
  
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) { 
      rowIndex = i + 1;
      break;
    }
  }
  
  if (rowIndex === -1) throw new Error("Data dengan ID " + id + " tidak ditemukan.");
  if(!dataObj.updated_at) dataObj.updated_at = new Date().toISOString();
  
  headers.forEach((header, colIndex) => {
    if (dataObj[header] !== undefined) {
      sheet.getRange(rowIndex, colIndex + 1).setValue(dataObj[header]);
    }
  });
  
  return { status: 'success', message: 'Data berhasil diupdate' };
}

function deleteData(tableName, id) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tableName);
  if (!sheet) throw new Error("Tabel tidak ditemukan: " + tableName);
  
  const data = sheet.getDataRange().getValues();
  let rowIndex = -1;
  
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) {
      rowIndex = i + 1;
      break;
    }
  }
  
  if (rowIndex === -1) throw new Error("Data tidak ditemukan.");
  sheet.deleteRow(rowIndex);
  
  return { status: 'success', message: 'Data berhasil dihapus' };
}
